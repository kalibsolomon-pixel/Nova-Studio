cmake_minimum_required(VERSION 3.24)
project(NovaQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL Concurrent Test)
add_library(nova-core STATIC src/core.cpp src/core.h)
target_link_libraries(nova-core PUBLIC Qt6::Gui)
target_include_directories(nova-core PUBLIC src)
add_executable(nova-qt WIN32 src/main.cpp src/window.cpp src/window.h src/canvas.cpp src/canvas.h src/cube.cpp src/cube.h src/assets.cpp src/assets.h src/metrics.cpp src/metrics.h)
target_link_libraries(nova-qt PRIVATE nova-core Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent)
if(MSVC)
  target_compile_options(nova-core PRIVATE /W4 /utf-8)
  target_compile_options(nova-qt PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(nova-qt-tests tests/core_test.cpp)
target_link_libraries(nova-qt-tests PRIVATE nova-core Qt6::Test)
add_test(NAME nova-core-tests COMMAND nova-qt-tests)
