cmake_minimum_required(VERSION 3.24)
project(NovaShutdownProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Concurrent)
if(NOT NOVA_BUILD)
  message(FATAL_ERROR "Set NOVA_BUILD to the verified Release build directory")
endif()
get_filename_component(NOVA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(nova-shutdown-probe probe.cpp)
foreach(part core app assets rendering ui models)
  target_include_directories(nova-shutdown-probe PRIVATE "${NOVA_SOURCE}/src/${part}")
endforeach()
foreach(lib workspace application core asset-index models diagnostics)
  target_link_libraries(nova-shutdown-probe PRIVATE "${NOVA_BUILD}/src/nova-${lib}.lib")
endforeach()
target_include_directories(nova-shutdown-probe PRIVATE "${NOVA_BUILD}/generated")
target_link_libraries(nova-shutdown-probe PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent Psapi)
if(MSVC)
  target_compile_options(nova-shutdown-probe PRIVATE /W4 /utf-8)
endif()
